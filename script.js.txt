let state = {
  lang: localStorage.getItem('ks_lang') || 'hi',
  theme: localStorage.getItem('ks_theme') || 'light',
  activeCat: 'home',
  activeSub: 'all',
  searchQ: '',
  selectedArticle: null
};

function t(key){ return STR[state.lang][key] || key; }

function init(){
  document.documentElement.setAttribute('data-theme', state.theme);
  renderLangSwitch();
  renderCategories();
  renderAll();
  bindEvents();
  updateTexts();
}

function renderLangSwitch(){
  const wrap = document.getElementById('langSwitch');
  if(!wrap) return;
  wrap.innerHTML = `<button data-lang="hi" class="${state.lang==='hi'?'active':''}">हिंदी</button><button data-lang="en" class="${state.lang==='en'?'active':''}">English</button>`;
  wrap.querySelectorAll('button').forEach(b=>{
    b.addEventListener('click',()=>{
      state.lang = b.dataset.lang;
      localStorage.setItem('ks_lang', state.lang);
      renderLangSwitch();
      renderAll();
      updateTexts();
    })
  })
}

function renderCategories(){
  const inner = document.getElementById('catInner');
  if(!inner) return;
  inner.innerHTML = CATEGORIES.map(c=>{
    const label = state.lang==='hi'?c.hi:c.en;
    return `<button class="cat-btn ${state.activeCat===c.key?'active':''}" data-cat="${c.key}">${label}</button>`
  }).join('');
  inner.querySelectorAll('.cat-btn').forEach(btn=>{
    btn.addEventListener('click',()=>{
      state.activeCat = btn.dataset.cat;
      state.activeSub = 'all';
      renderCategories();
      renderAll();
      // close drawer
      closeDrawer();
    })
  });
  // drawer categories
  const drawerGrid = document.getElementById('drawerCats');
  if(drawerGrid){
    drawerGrid.innerHTML = CATEGORIES.map(c=>{
      const label = state.lang==='hi'?c.hi:c.en;
      return `<button class="cat-btn ${state.activeCat===c.key?'active':''}" data-cat="${c.key}">${label}</button>`
    }).join('');
    drawerGrid.querySelectorAll('button').forEach(b=>{
      b.addEventListener('click',()=>{
        state.activeCat = b.dataset.cat;
        renderCategories();
        renderAll();
        closeDrawer();
        window.scrollTo({top:0, behavior:'smooth'});
      })
    })
  }
}

function filteredArticles(){
  let list = [...ARTICLES];
  const q = state.searchQ.trim().toLowerCase();
  if(state.activeCat!=='home' && state.activeCat!=='breaking'){
    list = list.filter(a=>a.category===state.activeCat || (state.activeCat==='education' && ['neet','jee'].includes(a.category)) );
  }
  if(state.activeSub!=='all'){
    list = list.filter(a=> (a.tags||[]).includes(state.activeSub) || a.category===state.activeSub);
  }
  if(q){
    list = list.filter(a=>{
      const hay = (a.titleHi + ' ' + a.titleEn + ' ' + a.category + ' ' + (a.tags||[]).join(' ')).toLowerCase();
      return hay.includes(q);
    })
  }
  return list;
}

function cardHTML(a){
  const title = state.lang==='hi'?a.titleHi:a.titleEn;
  const summary = state.lang==='hi'?a.summaryHi:a.summaryEn;
  return `<article class="card" data-id="${a.id}" tabindex="0" role="button" aria-label="${title}">
    <img loading="lazy" src="${a.image}" alt="${title}">
    <div class="card-body">
      <div class="card-cat">${a.category.toUpperCase()} • <span style="background:#fef3c7;color:#92400e;padding:2px 6px;border-radius:999px;font-size:9px">${t('demo')}</span></div>
      <div class="card-title">${title}</div>
      <div class="card-desc">${summary}</div>
      <div class="meta card-meta"><span>📅 ${a.publishedAt}</span><span>• ${a.author}</span></div>
    </div>
  </article>`;
}

function renderAll(){
  const list = filteredArticles();
  // Hero
  const heroMain = document.getElementById('heroMain');
  const heroSide = document.getElementById('heroSide');
  if(heroMain && heroSide){
    if(list.length>0){
      const main = list[0];
      const title = state.lang==='hi'?main.titleHi:main.titleEn;
      const summary = state.lang==='hi'?main.summaryHi:main.summaryEn;
      heroMain.innerHTML = `<img loading="lazy" src="${main.image}" alt="${title}"><div class="hero-overlay"><span class="badge-demo">⚠️ ${t('demo')} • ${t('liveUnavailable')}</span><br><span class="cat-label">${main.category.toUpperCase()}</span><h1 class="hero-title">${title}</h1><p class="hero-summary">${summary}</p><div class="meta dark"><span>📅 ${main.publishedAt}</span><span>✍️ ${main.author}</span><span style="margin-left:auto;background:white;color:#0f172a;padding:4px 10px;border-radius:999px;font-weight:800;font-size:11px">${t('readMore')} →</span></div></div>`;
      heroMain.onclick = ()=>openArticle(main.id);
      const sideList = list.slice(1,5);
      heroSide.innerHTML = sideList.map(a=>cardHTML(a)).join('');
      heroSide.querySelectorAll('.card').forEach(el=>{
        el.addEventListener('click',()=>openArticle(el.dataset.id));
        el.addEventListener('keydown',(e)=>{ if(e.key==='Enter') openArticle(el.dataset.id); })
      });
      heroMain.addEventListener('keydown',(e)=>{ if(e.key==='Enter') openArticle(main.id); });
      heroMain.setAttribute('tabindex','0');
      heroMain.setAttribute('role','button');
    }
  }

  // Sections
  renderSection('indiaSectionGrid','india');
  renderSection('worldSectionGrid','world');
  renderSection('rajasthanSectionGrid','rajasthan', true);
  renderSection('educationSectionGrid','education');
  renderSection('neetSectionGrid','neet', true);
  renderSection('resultsSectionGrid','results');
  renderSection('jobsSectionGrid','jobs');
  renderSection('techSectionGrid','technology');
  renderSection('sportsSectionGrid','sports');
  renderSection('entertainmentSectionGrid','entertainment');
  renderSection('factcheckSectionGrid','factcheck');

  // 4 cards grid below hero
  const grid4 = document.getElementById('grid4');
  if(grid4){
    const rest = list.slice(5,9);
    if(rest.length===0){
      grid4.innerHTML = `<div class="empty">${t('emptyCat')}</div>`;
    }else{
      grid4.innerHTML = rest.map(a=>cardHTML(a)).join('');
      grid4.querySelectorAll('.card').forEach(el=>el.addEventListener('click',()=>openArticle(el.dataset.id)));
    }
  }

  // Trending
  const trending = document.getElementById('trendingList');
  if(trending){
    const trendList = [...ARTICLES].slice(0,8);
    trending.innerHTML = trendList.map((a,i)=>{
      const title = state.lang==='hi'?a.titleHi:a.titleEn;
      return `<div class="trending-item" data-id="${a.id}" style="cursor:pointer"><div class="tr-num">0${i+1}</div><div><div style="font-size:13px;font-weight:700;line-height:1.3">${title}</div><div style="font-size:11px;color:var(--text-soft);margin-top:4px">${a.category} • ${a.publishedAt}</div></div></div>`
    }).join('');
    trending.querySelectorAll('.trending-item').forEach(el=>el.addEventListener('click',()=>openArticle(el.dataset.id)));
  }

  // Live updates
  const liveBox = document.getElementById('liveList');
  if(liveBox){
    liveBox.innerHTML = ARTICLES.slice(0,5).map(a=>{
      const title = state.lang==='hi'?a.titleHi:a.titleEn;
      return `<div class="live-item"><div class="live-dot"></div><div><div style="font-size:11px;color:var(--text-soft)">${a.publishedAt} • ${t('demo')}</div><div style="font-size:13px;font-weight:600;margin-top:2px">${title}</div></div></div>`
    }).join('');
  }

  // Search results info
  const searchInfo = document.getElementById('searchInfo');
  if(searchInfo){
    if(state.searchQ){
      searchInfo.textContent = `${list.length} results for "${state.searchQ}"`;
      searchInfo.classList.remove('hidden');
    }else{
      searchInfo.classList.add('hidden');
    }
  }
}

function renderSection(id, catKey, isProminent){
  const el = document.getElementById(id);
  if(!el) return;
  let list = ARTICLES.filter(a=>a.category===catKey);
  if(catKey==='education') list = ARTICLES.filter(a=>['education','neet','jee'].includes(a.category));
  // search also applies
  const q = state.searchQ.trim().toLowerCase();
  if(q){
    list = list.filter(a=>{
      const hay = (a.titleHi + ' ' + a.titleEn + ' ' + a.category + ' ' + (a.tags||[]).join(' ')).toLowerCase();
      return hay.includes(q);
    })
  }
  // subtab filter for rajasthan
  if(isProminent && state.activeSub!=='all' && catKey==='rajasthan'){
    list = list.filter(a=>a.tags.includes(state.activeSub));
  }
  if(list.length===0){
    el.innerHTML = `<div class="empty">${t('emptyCat')}</div>`;
    return;
  }
  el.innerHTML = list.slice(0,4).map(a=>cardHTML(a)).join('');
  el.querySelectorAll('.card').forEach(card=>card.addEventListener('click',()=>openArticle(card.dataset.id)));
}

  if(!a) return;
  state.selectedArticle = a;
  const overlay = document.getElementById('articleOverlay');
  const sheet = document.getElementById('articleSheet');
  const title = state.lang==='hi'?a.titleHi:a.titleEn;
  const summary = state.lang==='hi'?a.summaryHi:a.summaryEn;

  sheet.innerHTML = `

      <button id="closeArticle" class="icon-btn" aria-label="Close">✕</button>

function closeArticle(){
  const overlay = document.getElementById('articleOverlay');
  overlay.classList.remove('open');
  document.body.style.overflow='';
  state.selectedArticle=null;
}

function bindEvents(){
  // Search
  const searchInput = document.getElementById('searchInput');
  const searchBtn = document.getElementById('searchBtn');
  if(searchInput){
    searchInput.addEventListener('input',(e)=>{
      state.searchQ = e.target.value;
      renderAll();
    });
    searchInput.addEventListener('keydown',(e)=>{
      if(e.key==='Enter'){ state.searchQ = e.target.value; renderAll(); }
    });
  }
  if(searchBtn){
    searchBtn.addEventListener('click',()=>{
      state.searchQ = searchInput.value;
      renderAll();
      document.getElementById('mainContent')?.scrollIntoView({behavior:'smooth'});
    })
  }
  // Theme
  const themeBtn = document.getElementById('themeBtn');
  if(themeBtn){
    themeBtn.addEventListener('click',()=>{
      state.theme = state.theme==='light'?'dark':'light';
      localStorage.setItem('ks_theme', state.theme);
      document.documentElement.setAttribute('data-theme', state.theme);
      updateTexts();
    })
  }
  // Mobile drawer
  const hamb = document.getElementById('hamburger');
  const drawer = document.getElementById('mobileDrawer');
  const drawerBackdrop = document.getElementById('drawerBackdrop');
  const closeDrawerBtn = document.getElementById('closeDrawer');
  if(hamb) hamb.addEventListener('click',()=>{ drawer.classList.add('open'); });
  if(drawerBackdrop) drawerBackdrop.addEventListener('click', closeDrawer);
  if(closeDrawerBtn) closeDrawerBtn.addEventListener('click', closeDrawer);

  // Subtabs Rajasthan
  document.querySelectorAll('[data-subtab-group="rajasthan"] .subtab').forEach(btn=>{
    btn.addEventListener('click',()=>{
      document.querySelectorAll('[data-subtab-group="rajasthan"] .subtab').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      state.activeSub = btn.dataset.sub;
      renderSection('rajasthanSectionGrid','rajasthan', true);
    })
  });
  // Subtabs NEET
  document.querySelectorAll('[data-subtab-group="neet"] .subtab').forEach(btn=>{
    btn.addEventListener('click',()=>{
      document.querySelectorAll('[data-subtab-group="neet"] .subtab').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      // simple filter by tag for demo
      const grid = document.getElementById('neetSectionGrid');
      let list = ARTICLES.filter(a=>a.category==='neet' || a.tags.includes('neet'));
      if(btn.dataset.sub!=='all'){
        list = ARTICLES.filter(a=>a.tags.includes(btn.dataset.sub) || a.category===btn.dataset.sub);
      }
      if(list.length===0) grid.innerHTML = `<div class="empty">${t('emptyCat')}</div>`;
      else grid.innerHTML = list.slice(0,4).map(a=>cardHTML(a)).join('');
      grid.querySelectorAll('.card').forEach(card=>card.addEventListener('click',()=>openArticle(card.dataset.id)));
    })
  });

  // Overlay click to close
  const overlay = document.getElementById('articleOverlay');
  if(overlay){
    overlay.addEventListener('click',(e)=>{
      if(e.target===overlay) closeArticle();
    })
  }
  document.addEventListener('keydown',(e)=>{
    if(e.key==='Escape') { closeArticle(); closeDrawer(); }
  })
}

function closeDrawer(){
  document.getElementById('mobileDrawer')?.classList.remove('open');
}

function updateTexts(){
  // placeholders
  const searchInput = document.getElementById('searchInput');
  if(searchInput) searchInput.placeholder = t('searchPlaceholder');
  const themeBtn = document.getElementById('themeBtn');
  if(themeBtn) themeBtn.textContent = state.theme==='light'?'🌙 '+t('dark'):'☀️ '+t('light');
  // static labels
  document.querySelectorAll('[data-i18n]').forEach(el=>{
    const key = el.dataset.i18n;
    if(STR[state.lang][key]) el.textContent = t(key);
  });
  // footer etc already dynamic via render
}
